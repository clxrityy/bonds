1use bonds_cli::ui;
2use bonds_core::{BondError, BondManager};
3
4/// Remove command handler, which deletes a bond by its ID or name. The command retrieves the bond, deletes it using the bond manager's `delete_bond` method, and provides user feedback on the success of the operation, including details of the removed bond's source and target paths. If the `with_target` flag is set, it also deletes the target directory associated with the bond.
5/// **Example usage:**
6/// ```bash
7/// bond remove <id | name> [--with-target]
8/// ```
9pub fn cmd_remove(
10    manager: &BondManager,
11    id: &str,
12    with_target: bool,
13    dry_run: bool,
14    verbose: bool,
15) -> Result<(), BondError> {
16    let bond = manager.get_bond(id)?;
17
18    if verbose {
19        ui::debug(format!("Removing bond with ID: {}", id));
20        if with_target {
21            ui::debug(format!(
22                "Target directory will also be deleted: {}",
23                bond.target().display()
24            ));
25        } else {
26            ui::debug("The target directory will NOT be deleted.");
27        }
28    }
29
30    if dry_run {
31        ui::warning("Dry run enabled: no changes will be made.");
32        ui::info(format!("Would remove bond with ID: {}", id));
33        if with_target {
34            ui::info("Would also delete the target directory.");
35        } else {
36            ui::info("Would NOT delete the target directory.");
37        }
38        return Ok(());
39    }
40
41    let bond = manager.delete_bond(id, with_target)?;
42    ui::status_ok("✓ SUCCESS");
43    ui::success("Bond removed:");
44    ui::id(format!("   {}", bond.id()));
45    ui::dim(format!("   {}", bond.source().display()));
46    ui::normal("    ⤡ ");
47    ui::dim(format!("    {}\n", bond.target().display()));
48
49    if with_target {
50        ui::info("  Target directory also deleted.");
51    }
52
53    Ok(())
54}

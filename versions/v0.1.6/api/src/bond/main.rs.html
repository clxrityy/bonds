1//! The main entry point for the bonds CLI application. It parses command-line arguments, initializes the bond manager, and dispatches commands to their respective handlers.
2//!     The CLI supports various commands for managing symlinks, including adding, listing, updating, and removing bonds, as well as configuration and metadata management.
3//! - Error handling is centralized, ensuring that any issues encountered during command execution are reported in a consistent manner.
4//! - The application relies on the [`bonds-core`](https://crates.io/crates/bonds-core) crate for core logic and uses the `clap` crate for argument parsing. The user interface is handled through the `ui` module, which provides formatted output for errors and other messages.
5
6use bonds_cli::args::{Cli, Commands, ConfigAction, MetadataAction};
7use commands::{
8    cmd_add, cmd_config_get, cmd_config_set, cmd_info, cmd_list, cmd_metadata_get,
9    cmd_metadata_remove, cmd_metadata_set, cmd_migrate, cmd_remove, cmd_update,
10};

12/// The `commands` module contains the implementation of all the command handlers for the CLI application. Each command corresponds to a specific action that can be performed on the bonds, such as adding a new bond, listing existing bonds, updating bond information, and managing metadata. The command handlers interact with the `BondManager` from the `bonds-core` crate to perform the necessary operations on the underlying data store. This modular structure allows for clean separation of concerns and makes it easier to maintain and extend the CLI functionality in the future.
13mod commands;
14/// The `ui` module provides functions for formatting error messages and other user-facing output in a consistent and visually appealing way. It is used throughout the CLI application to ensure that all messages are presented in a clear and user-friendly manner.
15mod ui;
16
17use clap::Parser;
18
19fn main() {
20    let cli = Cli::parse();
21
22    let result = match cli.command {
23        Commands::Config { action } => match action {
24            ConfigAction::Get { key } => cmd_config_get(&key),
25            ConfigAction::Set { key, value } => cmd_config_set(&key, &value),
26        },
27        cmd => {
28            // Only init DB for commands that need it.
29            let manager = match bonds_core::BondManager::new(cli.db) {
30                Ok(m) => m,
31                Err(e) => {
32                    // Use the same color path as all other errors.
33                    eprintln!(
34                        "{}",
35                        ui::format_context_error("Failed to initialize bond manager", &e)
36                    );
37                    std::process::exit(1);
38                }
39            };
40
41            match cmd {
42                Commands::Add {
43                    source,
44                    target,
45                    contents,
46                    name,
47                } => cmd_add(&manager, source, target, contents, name),
48                Commands::List => cmd_list(&manager),
49                Commands::Info { id } => cmd_info(&manager, &id),
50                Commands::Remove { id, with_target } => cmd_remove(&manager, &id, with_target),
51                Commands::Config { .. } => unreachable!(),
52                Commands::Update {
53                    id,
54                    source,
55                    target,
56                    name,
57                } => cmd_update(&manager, &id, source, target, name),
58                Commands::Migrate { id, dest } => cmd_migrate(&manager, &id, dest),
59                Commands::Metadata { action } => match action {
60                    MetadataAction::Get { id, key } => {
61                        // None key means "print all metadata"
62                        cmd_metadata_get(&manager, &id, key.as_deref())
63                    }
64                    MetadataAction::Set { id, key, value } => {
65                        // Upsert metadata key/value on the selected bond
66                        cmd_metadata_set(&manager, &id, &key, &value)
67                    }
68                    MetadataAction::Remove { id, key } => {
69                        // Remove a single key from metadata map
70                        cmd_metadata_remove(&manager, &id, &key)
71                    }
72                },
73            }
74        }
75    };
76
77    if let Err(e) = result {
78        // One place for every command failure.
79        ui::format_error(&e);
80        std::process::exit(1);
81    }
82}

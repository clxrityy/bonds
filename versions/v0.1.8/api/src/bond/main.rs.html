1//! The main entry point for the bonds CLI application. It parses command-line arguments, initializes the bond manager, and dispatches commands to their respective handlers.
2//!     The CLI supports various commands for managing symlinks, including adding, listing, updating, and removing bonds, as well as configuration and metadata management.
3//! - Error handling is centralized, ensuring that any issues encountered during command execution are reported in a consistent manner.
4//! - The application relies on the [`bonds-core`](https://crates.io/crates/bonds-core) crate for core logic and uses the `clap` crate for argument parsing. The user interface is handled through the `ui` module, which provides formatted output for errors and other messages.
5
6use bonds_cli::args::{Cli, Commands, ConfigAction, MetadataAction};
7use commands::{
8    cmd_add, cmd_config_get, cmd_config_set, cmd_info, cmd_list, cmd_metadata_get,
9    cmd_metadata_remove, cmd_metadata_set, cmd_migrate, cmd_remove, cmd_update,
10};
11
12/// The `commands` module contains the implementation of all the command handlers for the CLI application. Each command corresponds to a specific action that can be performed on the bonds, such as adding a new bond, listing existing bonds, updating bond information, and managing metadata. The command handlers interact with the `BondManager` from the `bonds-core` crate to perform the necessary operations on the underlying data store. This modular structure allows for clean separation of concerns and makes it easier to maintain and extend the CLI functionality in the future.
13mod commands;
14/// The `ui` module provides functions for formatting error messages and other user-facing output in a consistent and visually appealing way. It is used throughout the CLI application to ensure that all messages are presented in a clear and user-friendly manner.
15mod ui;
16
17use clap::Parser;
18
19fn main() {
20    let cli = Cli::parse();
21
22    // Branded default landing output when no subcommand is provided.
23    let Some(cmd) = cli.command else {
24        ui::landing(env!("CARGO_PKG_VERSION"));
25        return;
26    };
27
28    let result = match cmd {
29        Commands::Config { action } => match action {
30            ConfigAction::Get { key } => cmd_config_get(&key),
31            ConfigAction::Set { key, value } => cmd_config_set(&key, &value),
32        },
33        Commands::Add {
34            source,
35            target,
36            name,
37            flags,
38        } => {
39            // Only init DB for commands that need it.
40            let manager = match bonds_core::BondManager::new(cli.db) {
41                Ok(m) => m,
42                Err(e) => {
43                    // Use the same color path as all other errors.
44                    eprintln!(
45                        "{}",
46                        ui::format_context_error("Failed to initialize bond manager", &e)
47                    );
48                    std::process::exit(1);
49                }
50            };
51            cmd_add(&manager, source, target, name, flags.dry_run, flags.verbose)
52        }
53        Commands::List => {
54            let manager = match bonds_core::BondManager::new(cli.db) {
55                Ok(m) => m,
56                Err(e) => {
57                    eprintln!(
58                        "{}",
59                        ui::format_context_error("Failed to initialize bond manager", &e)
60                    );
61                    std::process::exit(1);
62                }
63            };
64            cmd_list(&manager)
65        }
66        Commands::Info { id } => {
67            let manager = match bonds_core::BondManager::new(cli.db) {
68                Ok(m) => m,
69                Err(e) => {
70                    eprintln!(
71                        "{}",
72                        ui::format_context_error("Failed to initialize bond manager", &e)
73                    );
74                    std::process::exit(1);
75                }
76            };
77            cmd_info(&manager, &id)
78        }
79        Commands::Remove {
80            id,
81            with_target,
82            flags,
83        } => {
84            let manager = match bonds_core::BondManager::new(cli.db) {
85                Ok(m) => m,
86                Err(e) => {
87                    eprintln!(
88                        "{}",
89                        ui::format_context_error("Failed to initialize bond manager", &e)
90                    );
91                    std::process::exit(1);
92                }
93            };
94            cmd_remove(&manager, &id, with_target, flags.dry_run, flags.verbose)
95        }
96        Commands::Update {
97            id,
98            source,
99            target,
100            name,
101            flags,
102        } => {
103            let manager = match bonds_core::BondManager::new(cli.db) {
104                Ok(m) => m,
105                Err(e) => {
106                    eprintln!(
107                        "{}",
108                        ui::format_context_error("Failed to initialize bond manager", &e)
109                    );
110                    std::process::exit(1);
111                }
112            };
113            cmd_update(
114                &manager,
115                &id,
116                source,
117                target,
118                name,
119                flags.dry_run,
120                flags.verbose,
121            )
122        }
123        Commands::Migrate { id, dest, flags } => {
124            let manager = match bonds_core::BondManager::new(cli.db) {
125                Ok(m) => m,
126                Err(e) => {
127                    eprintln!(
128                        "{}",
129                        ui::format_context_error("Failed to initialize bond manager", &e)
130                    );
131                    std::process::exit(1);
132                }
133            };
134            cmd_migrate(&manager, &id, dest, flags.dry_run, flags.verbose)
135        }
136        Commands::Metadata { action } => {
137            let manager = match bonds_core::BondManager::new(cli.db) {
138                Ok(m) => m,
139                Err(e) => {
140                    eprintln!(
141                        "{}",
142                        ui::format_context_error("Failed to initialize bond manager", &e)
143                    );
144                    std::process::exit(1);
145                }
146            };
147            match action {
148                MetadataAction::Get { id, key } => {
149                    // None key means "print all metadata"
150                    cmd_metadata_get(&manager, &id, key.as_deref())
151                }
152                MetadataAction::Set { id, key, value } => {
153                    // Upsert metadata key/value on the selected bond
154                    cmd_metadata_set(&manager, &id, &key, &value)
155                }
156                MetadataAction::Remove { id, key } => {
157                    // Remove a single key from metadata map
158                    cmd_metadata_remove(&manager, &id, &key)
159                }
160            }
161        }
162    };
163
164    if let Err(e) = result {
165        // One place for every command failure.
166        ui::format_error(&e);
167        std::process::exit(1);
168    }
169}

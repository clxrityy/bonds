1use bonds_core::{Bond, BondError, BondManager};
2use serde::Serialize;
3use std::path::PathBuf;
4
5#[derive(Debug, Clone, Serialize)]
6#[serde(rename_all = "camelCase")]
7struct BondListItem {
8    id: String,
9    name: Option<String>,
10    source: String,
11    target: String,
12    created_at: String,
13    status: String,
14    metadata_count: usize,
15}
16
17fn load_bonds(db_path: Option<PathBuf>) -> Result<Vec<BondListItem>, BondError> {
18    let manager = BondManager::new(db_path)?;
19    let bonds = manager.list_bonds()?;
20    Ok(bonds.into_iter().map(map_bond).collect())
21}
22
23fn map_bond(bond: Bond) -> BondListItem {
24    BondListItem {
25        id: bond.id().to_string(),
26        name: bond.name().map(str::to_owned),
27        source: bond.source().display().to_string(),
28        target: bond.target().display().to_string(),
29        created_at: bond.created_at_rfc3339(),
30        status: bond_status(&bond),
31        metadata_count: bond.metadata().map_or(0, |m| m.len()),
32    }
33}
34
35fn bond_status(bond: &Bond) -> String {
36    let target_exists = bond.target().exists();
37    let is_symlink = bond
38        .target()
39        .symlink_metadata()
40        .map(|meta| meta.file_type().is_symlink())
41        .unwrap_or(false);
42
43    match (target_exists, is_symlink) {
44        (true, true) => "healthy".to_string(),
45        (true, false) => "warning".to_string(),
46        (false, _) => "broken".to_string(),
47    }
48}

50#[tauri::command]
51fn list_bonds(db_path: Option<String>) -> Result<Vec<BondListItem>, String> {
52    load_bonds(db_path.map(PathBuf::from)).map_err(|err| err.to_string())
53}
54
55#[cfg_attr(mobile, tauri::mobile_entry_point)]
56pub fn run() {
57    tauri::Builder::default()
58        .invoke_handler(tauri::generate_handler![list_bonds])
59        .run(tauri::generate_context!())
60        .expect("failed to run Bonds desktop app");
61}
